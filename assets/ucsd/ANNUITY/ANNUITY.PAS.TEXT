PROGRAM ANNUITY;
{ Apple Pascal 1.3 source. All money is stored as integer cents. }
{ This distribution is plain ASCII, not an encoded UCSD text file. }
{$I-}
TYPE
  MONEY = INTEGER[20];
  LINE = STRING[80];
  WIDELINE = STRING[255];
  DATE = RECORD Y, M, D: INTEGER END;
  PAYMENT = RECORD WHEN: DATE; AMOUNT: MONEY END;
  LOAN = RECORD
    NAME, DESCR: LINE;
    CAPITAL, FIXED: MONEY;
    RATE, GIFT, TERM, MODE: INTEGER;
    START: DATE;
    COUNT: INTEGER;
    EXTRA: ARRAY[1..100] OF PAYMENT
  END;
  TOTALS = RECORD INT, REPAY, NET, EXTRA, GIFT: MONEY END;
VAR
  L, OLD: LOAN;
  NAMES: ARRAY[1..50] OF LINE;
  NLOANS, ACTIVE, IOERR, LINES: INTEGER;
  F, OUTFILE: TEXT;
  S: LINE;
  QUIT, BACK, OK, TOFILE: BOOLEAN;
  LIMIT: MONEY;

PROCEDURE NUM(N: MONEY; VAR T: LINE);
BEGIN STR(N,T) END;

PROCEDURE PAD(N: INTEGER; VAR T: LINE);
BEGIN NUM(N,T); IF N < 10 THEN T := CONCAT('0',T) END;

PROCEDURE FMT(V: MONEY; VAR T: LINE);
VAR NEG: BOOLEAN;
BEGIN
  NEG := V < 0; IF NEG THEN V := -V;
  STR(V,T);
  WHILE LENGTH(T) < 3 DO T := CONCAT('0',T);
  INSERT('.',T,LENGTH(T)-1);
  IF NEG THEN T := CONCAT('-',T)
END;

PROCEDURE FMTDATE(V: DATE; VAR T: LINE);
VAR A,B,C: LINE;
BEGIN
  PAD(V.D,A); PAD(V.M,B); NUM(V.Y,C);
  T := CONCAT(A,'-',B,'-',C)
END;

FUNCTION DAYS(Y,M: INTEGER): INTEGER;
BEGIN
  CASE M OF
    4,6,9,11: DAYS := 30;
    2: IF (Y MOD 4 = 0) AND
          ((Y MOD 100 <> 0) OR (Y MOD 400 = 0))
       THEN DAYS := 29 ELSE DAYS := 28;
    1,3,5,7,8,10,12: DAYS := 31
  END
END;

FUNCTION BEFORE(A,B: DATE): BOOLEAN;
BEGIN
  BEFORE := (A.Y < B.Y) OR ((A.Y = B.Y) AND (A.M < B.M)) OR
    ((A.Y = B.Y) AND (A.M = B.M) AND (A.D < B.D))
END;

PROCEDURE NEXTMON(VAR V: DATE);
BEGIN V.D := 1; V.M := V.M+1;
  IF V.M = 13 THEN BEGIN V.M := 1; V.Y := V.Y+1 END
END;

{ Parse unsigned amounts; decimal point or comma, no grouping. }
FUNCTION PARSE(T: LINE; SCALE: INTEGER; VAR V: MONEY): BOOLEAN;
VAR I, DIG, DECIM: INTEGER; DOT, GOOD: BOOLEAN;
BEGIN
  V := 0; DOT := FALSE; GOOD := LENGTH(T) > 0; DECIM := 0;
  FOR I := 1 TO LENGTH(T) DO
    IF (T[I] >= '0') AND (T[I] <= '9') THEN BEGIN
      DIG := ORD(T[I])-ORD('0');
      IF DOT THEN DECIM := DECIM+1;
      IF (DECIM > SCALE) OR (V > LIMIT) THEN GOOD := FALSE
      ELSE V := V*10+DIG
    END ELSE
      IF ((T[I] = '.') OR (T[I] = ',')) AND NOT DOT
         AND (SCALE > 0) AND (I > 1) AND (I < LENGTH(T))
      THEN DOT := TRUE ELSE GOOD := FALSE;
  WHILE DECIM < SCALE DO BEGIN V := V*10; DECIM := DECIM+1 END;
  PARSE := GOOD AND (V <= LIMIT)
END;

FUNCTION ASKVAL(PROMPT: LINE; SCALE: INTEGER;
                LOW,HIGH: MONEY; VAR V: MONEY): BOOLEAN;
VAR T: LINE; DONE, GOOD: BOOLEAN;
BEGIN
  DONE := FALSE; GOOD := FALSE;
  REPEAT
    WRITE(PROMPT); READLN(T);
    IF (T = 'q') OR (T = 'Q') THEN DONE := TRUE
    ELSE BEGIN
      GOOD := PARSE(T,SCALE,V);
      IF GOOD THEN GOOD := (V >= LOW) AND (V <= HIGH);
      DONE := GOOD;
      IF NOT DONE THEN WRITELN('Invalid value. Use Q to cancel.')
    END
  UNTIL DONE;
  ASKVAL := GOOD
END;

FUNCTION ASKINT(PROMPT: LINE; LOW,HIGH: INTEGER;
                VAR V: INTEGER): BOOLEAN;
VAR B: MONEY; GOOD: BOOLEAN;
BEGIN
  GOOD := ASKVAL(PROMPT,0,LOW,HIGH,B);
  IF GOOD THEN V := TRUNC(B);
  ASKINT := GOOD
END;

FUNCTION GETDATE(T: LINE; VAR V: DATE): BOOLEAN;
VAR A,B,C: MONEY; GOOD: BOOLEAN;
BEGIN
  GOOD := LENGTH(T) = 10;
  IF GOOD THEN GOOD := (T[3] = '-') AND (T[6] = '-');
  IF GOOD THEN GOOD := PARSE(COPY(T,1,2),0,A) AND
    PARSE(COPY(T,4,2),0,B) AND PARSE(COPY(T,7,4),0,C);
  IF GOOD THEN GOOD := (A >= 1) AND (A <= 31) AND
    (B >= 1) AND (B <= 12) AND (C >= 1900) AND (C <= 9998);
  IF GOOD THEN BEGIN
    V.D := TRUNC(A); V.M := TRUNC(B); V.Y := TRUNC(C);
    GOOD := V.D <= DAYS(V.Y,V.M)
  END;
  GETDATE := GOOD
END;

FUNCTION ASKDATE(PROMPT: LINE; VAR V: DATE): BOOLEAN;
VAR T: LINE; GOOD, CANCEL: BOOLEAN;
BEGIN
  REPEAT
    WRITE(PROMPT); READLN(T); CANCEL := (T = 'q') OR (T = 'Q');
    GOOD := GETDATE(T,V);
    IF NOT GOOD AND NOT CANCEL THEN WRITELN('Use DD-MM-YYYY or Q.')
  UNTIL GOOD OR CANCEL;
  ASKDATE := GOOD
END;

PROCEDURE FNAME(PREFIX: LINE; N: INTEGER; VAR T: LINE);
BEGIN
  NUM(N,T); WHILE LENGTH(T) < 4 DO T := CONCAT('0',T);
  T := CONCAT(PREFIX,T,'.TEXT')
END;

PROCEDURE EMIT(T: WIDELINE);
BEGIN
  IF TOFILE THEN BEGIN
    WRITELN(OUTFILE,T);
    IF IORESULT <> 0 THEN IOERR := 1
  END ELSE BEGIN
    WRITELN(T); LINES := LINES+1;
    IF LINES = 20 THEN BEGIN
      WRITE('Press Return: '); READLN(S); LINES := 0
    END
  END
END;

PROCEDURE CLEAR(VAR T: TOTALS);
BEGIN T.INT := 0; T.REPAY := 0; T.NET := 0;
  T.EXTRA := 0; T.GIFT := 0
END;

PROCEDURE ADD(VAR A: TOTALS; B: TOTALS);
BEGIN
  A.INT := A.INT+B.INT; A.REPAY := A.REPAY+B.REPAY;
  A.NET := A.NET+B.NET; A.EXTRA := A.EXTRA+B.EXTRA;
  A.GIFT := A.GIFT+B.GIFT
END;

{ Vertical rows remain readable on a 40-column display. }
PROCEDURE PUTM(LABELS: LINE; V: MONEY);
VAR T: LINE;
BEGIN FMT(V,T); EMIT(CONCAT(LABELS,T)) END;

PROCEDURE ROW(LABELS: LINE; T: TOTALS; BALANCE: MONEY);
BEGIN
  EMIT(LABELS);
  PUTM('  Interest: ',T.INT);
  PUTM('  Principal: ',T.REPAY);
  PUTM('  Net principal: ',T.NET);
  PUTM('  Extra: ',T.EXTRA);
  PUTM('  Balance: ',BALANCE);
  PUTM('  Gift: ',T.GIFT)
END;

PROCEDURE SORT;
VAR I,J: INTEGER; T: PAYMENT; SWAP: BOOLEAN;
BEGIN
  FOR I := 2 TO L.COUNT DO BEGIN
    J := I;
    WHILE J > 1 DO BEGIN
      SWAP := BEFORE(L.EXTRA[J].WHEN,L.EXTRA[J-1].WHEN);
      IF NOT SWAP THEN
        IF NOT BEFORE(L.EXTRA[J-1].WHEN,L.EXTRA[J].WHEN) THEN
          SWAP := L.EXTRA[J].AMOUNT < L.EXTRA[J-1].AMOUNT;
      IF SWAP THEN BEGIN
        T := L.EXTRA[J]; L.EXTRA[J] := L.EXTRA[J-1];
        L.EXTRA[J-1] := T; J := J-1
      END ELSE J := 1
    END
  END
END;

PROCEDURE HEADER;
VAR T: LINE;
BEGIN
  EMIT('ANNUITY - LOAN SCHEDULE'); EMIT(L.NAME);
  IF LENGTH(L.DESCR) > 0 THEN EMIT(L.DESCR);
  PUTM('Capital: EUR ',L.CAPITAL);
  FMTDATE(L.START,T); EMIT(CONCAT('Start: ',T));
  PUTM('Annual interest %: ',L.RATE);
  PUTM('Annual gift %: ',L.GIFT);
  CASE L.MODE OF
    1: EMIT('Mode: constant principal repayment');
    2: EMIT('Mode: interest only');
    3: PUTM('Monthly total incl. interest: ',L.FIXED)
  END;
  EMIT('All amounts are EUR.'); EMIT('')
END;

PROCEDURE CHARGE(WEIGHT: MONEY; RATE,NDAYS: INTEGER;
                 VAR VALUE: MONEY);
VAR DEN: MONEY;
BEGIN
  DEN := 1200; DEN := DEN*100*NDAYS;
  VALUE := (WEIGHT*RATE+DEN DIV 2) DIV DEN
END;

{ Show and validation runs use exactly the same calculation. }
FUNCTION DATESOK: BOOLEAN;
VAR LAST: DATE; I: INTEGER; GOOD: BOOLEAN;
BEGIN
  LAST := L.START; LAST.D := 1;
  FOR I := 1 TO L.TERM DO NEXTMON(LAST);
  GOOD := TRUE;
  FOR I := 1 TO L.COUNT DO BEGIN
    IF BEFORE(L.EXTRA[I].WHEN,L.START) THEN GOOD := FALSE;
    IF L.MODE <> 3 THEN
      IF NOT BEFORE(L.EXTRA[I].WHEN,LAST) THEN GOOD := FALSE
  END;
  IF NOT GOOD THEN
    WRITELN('Extra date is outside this payment mode term.');
  DATESOK := GOOD
END;

PROCEDURE SCHEDULE(SHOW: BOOLEAN; VAR VALID: BOOLEAN);
VAR
  CUR: DATE; BAL, BASE, WEIGHT, AMT: MONEY;
  MT,YT,GT: TOTALS;
  K,PREV,ND,SPAN: INTEGER; MONTHS: MONEY;
  DONE, MORE: BOOLEAN; ERR,A,B: LINE;
BEGIN
  VALID := DATESOK; IF NOT VALID THEN EXIT(SCHEDULE);
  SORT; BAL := L.CAPITAL;
  BASE := (L.CAPITAL+L.TERM DIV 2) DIV L.TERM;
  CUR := L.START; CUR.D := 1; K := 1; MONTHS := 0;
  VALID := TRUE; DONE := FALSE; ERR := '';
  CLEAR(YT); CLEAR(GT);
  IF SHOW THEN HEADER;
  REPEAT
    CLEAR(MT); WEIGHT := 0; ND := DAYS(CUR.Y,CUR.M); PREV := 1;
    IF MONTHS = 0 THEN PREV := L.START.D;
    MORE := K <= L.COUNT;
    WHILE MORE DO BEGIN
      IF (L.EXTRA[K].WHEN.Y = CUR.Y) AND
         (L.EXTRA[K].WHEN.M = CUR.M) THEN BEGIN
        SPAN := L.EXTRA[K].WHEN.D-PREV;
        WEIGHT := WEIGHT+BAL*SPAN;
        AMT := L.EXTRA[K].AMOUNT; IF AMT > BAL THEN AMT := BAL;
        BAL := BAL-AMT; MT.EXTRA := MT.EXTRA+AMT;
        IF SHOW AND (AMT > 0) THEN
        BEGIN
          FMTDATE(L.EXTRA[K].WHEN,A); FMT(AMT,B);
          EMIT(CONCAT('Extra ',A,': ',B))
        END;
        PREV := L.EXTRA[K].WHEN.D; K := K+1
      END ELSE MORE := FALSE;
      IF K > L.COUNT THEN MORE := FALSE
    END;
    WEIGHT := WEIGHT+BAL*(ND+1-PREV);
    CHARGE(WEIGHT,L.RATE,ND,MT.INT);
    CHARGE(WEIGHT,L.GIFT,ND,MT.GIFT);
    MONTHS := MONTHS+1;
    CASE L.MODE OF
      1: BEGIN
        MT.REPAY := BASE;
        IF MONTHS = L.TERM THEN MT.REPAY := BAL
      END;
      2: MT.REPAY := 0;
      3: BEGIN
        IF L.FIXED < MT.INT THEN BEGIN
          VALID := FALSE; ERR := 'Monthly total does not cover interest.'
        END ELSE MT.REPAY := L.FIXED-MT.INT;
        IF VALID AND (MT.REPAY = 0) AND (BAL > 0)
           AND (K > L.COUNT) THEN
        BEGIN VALID := FALSE; ERR := 'Monthly total cannot repay this loan.' END
      END
    END;
    IF MT.REPAY > BAL THEN MT.REPAY := BAL;
    IF VALID THEN BEGIN
      BAL := BAL-MT.REPAY; MT.NET := MT.REPAY-MT.GIFT;
      ADD(YT,MT); ADD(GT,MT);
      DONE := (BAL = 0) OR ((L.MODE <> 3) AND (MONTHS = L.TERM));
      IF SHOW THEN BEGIN
        PAD(CUR.M,A); NUM(CUR.Y,B); A := CONCAT(A,'-',B);
        ROW(A,MT,BAL)
      END;
      IF (CUR.M = 12) OR DONE THEN BEGIN
        IF SHOW THEN BEGIN
          NUM(CUR.Y,A); A := CONCAT('YEAR ',A,' TOTAL');
          ROW(A,YT,BAL); EMIT('')
        END;
        CLEAR(YT)
      END;
      NEXTMON(CUR);
      IF NOT DONE AND (CUR.Y > 9998) THEN BEGIN
        VALID := FALSE; ERR := 'Schedule exceeds supported year 9998.'
      END
    END
  UNTIL DONE OR NOT VALID;
  IF NOT VALID THEN WRITELN(ERR)
  ELSE IF SHOW THEN BEGIN
    ROW('GRAND TOTAL',GT,BAL);
    NUM(MONTHS,A); EMIT(CONCAT('Months shown: ',A));
    IF BAL > 0 THEN EMIT('Outstanding principal remains payable.');
    EMIT('Net principal = principal minus gift.');
    EMIT('Extra payments are capped at the balance.');
    EMIT('Gift rate is not a tax exemption decision.')
  END
END;

PROCEDURE READLINE(VAR T: LINE; VAR GOOD: BOOLEAN);
BEGIN
  T := '';
  IF GOOD THEN BEGIN
    IF EOF(F) THEN GOOD := FALSE ELSE BEGIN
      READLN(F,T); IF IORESULT <> 0 THEN GOOD := FALSE
    END
  END
END;

PROCEDURE READNUM(VAR V: MONEY; VAR GOOD: BOOLEAN);
VAR T: LINE;
BEGIN
  READLINE(T,GOOD);
  IF GOOD THEN GOOD := PARSE(T,0,V)
END;

PROCEDURE READINT(VAR V: INTEGER; LOW,HIGH: INTEGER;
                  VAR GOOD: BOOLEAN);
VAR B: MONEY;
BEGIN
  READNUM(B,GOOD);
  IF GOOD THEN GOOD := (B >= LOW) AND (B <= HIGH);
  IF GOOD THEN V := TRUNC(B)
END;

PROCEDURE LOADLOAN(N: INTEGER; VAR GOOD: BOOLEAN);
VAR T: LINE; I: INTEGER;
BEGIN
  FNAME('L',N,T); RESET(F,T); GOOD := IORESULT = 0;
  IF NOT GOOD THEN WRITELN('Cannot open loan file.') ELSE BEGIN
    READLINE(T,GOOD); IF T <> 'ANNUITY 1' THEN GOOD := FALSE;
    READLINE(L.NAME,GOOD); READLINE(L.DESCR,GOOD);
    READNUM(L.CAPITAL,GOOD); READNUM(L.FIXED,GOOD);
    READINT(L.RATE,0,10000,GOOD); READINT(L.GIFT,0,10000,GOOD);
    READINT(L.TERM,1,1200,GOOD); READINT(L.MODE,1,3,GOOD);
    READLINE(T,GOOD); IF GOOD THEN GOOD := GETDATE(T,L.START);
    READINT(L.COUNT,0,100,GOOD);
    IF GOOD THEN FOR I := 1 TO L.COUNT DO BEGIN
      READLINE(T,GOOD);
      IF GOOD THEN GOOD := GETDATE(T,L.EXTRA[I].WHEN);
      READNUM(L.EXTRA[I].AMOUNT,GOOD);
      IF GOOD THEN GOOD := (L.EXTRA[I].AMOUNT > 0) AND
        NOT BEFORE(L.EXTRA[I].WHEN,L.START)
    END;
    CLOSE(F); IF IORESULT <> 0 THEN GOOD := FALSE;
    IF GOOD THEN GOOD := (L.CAPITAL > 0) AND (L.GIFT <= L.RATE)
      AND ((L.MODE <> 3) OR (L.FIXED > 0));
    IF GOOD THEN SCHEDULE(FALSE,GOOD);
    IF NOT GOOD THEN WRITELN('Invalid loan file; no changes made.')
  END
END;

PROCEDURE SAVELN(T: LINE);
BEGIN WRITELN(F,T); IF IORESULT <> 0 THEN IOERR := 1 END;

PROCEDURE SAVENUM(V: MONEY);
VAR T: LINE;
BEGIN STR(V,T); SAVELN(T) END;

PROCEDURE WRITELOAN(PREFIX: LINE; VAR GOOD: BOOLEAN);
VAR I: INTEGER; T: LINE;
BEGIN
  IOERR := 0; FNAME(PREFIX,ACTIVE,T); REWRITE(F,T);
  GOOD := IORESULT = 0;
  IF GOOD THEN BEGIN
    SAVELN('ANNUITY 1'); SAVELN(L.NAME); SAVELN(L.DESCR);
    SAVENUM(L.CAPITAL); SAVENUM(L.FIXED);
    SAVENUM(L.RATE); SAVENUM(L.GIFT); SAVENUM(L.TERM); SAVENUM(L.MODE);
    FMTDATE(L.START,T); SAVELN(T); SAVENUM(L.COUNT);
    FOR I := 1 TO L.COUNT DO BEGIN
      FMTDATE(L.EXTRA[I].WHEN,T); SAVELN(T); SAVENUM(L.EXTRA[I].AMOUNT)
    END;
    IF IOERR = 0 THEN CLOSE(F,LOCK) ELSE CLOSE(F);
    IF IORESULT <> 0 THEN IOERR := 1;
    GOOD := IOERR = 0
  END;
  IF NOT GOOD THEN WRITELN('Cannot save loan. Check disk space.')
END;

PROCEDURE COMMIT;
VAR NEWLOAN: LOAN; GOOD: BOOLEAN;
BEGIN
  SCHEDULE(FALSE,GOOD);
  IF GOOD THEN BEGIN
    NEWLOAN := L; L := OLD; WRITELOAN('B',GOOD); L := NEWLOAN;
    IF GOOD THEN WRITELOAN('L',GOOD)
  END;
  IF GOOD THEN WRITELN('Saved.') ELSE BEGIN
    L := OLD; WRITELN('Change not saved. Previous data retained.')
  END
END;

PROCEDURE INDEXIN;
VAR GOOD: BOOLEAN; I: INTEGER;
BEGIN
  NLOANS := 0; RESET(F,'LOANS.TEXT'); IOERR := IORESULT;
  IF IOERR = 10 THEN WRITELN('No loan index yet.')
  ELSE IF IOERR <> 0 THEN BEGIN
    WRITELN('Cannot read loan index.'); QUIT := TRUE
  END ELSE BEGIN
    GOOD := TRUE; READLINE(S,GOOD);
    IF S <> 'ANNUITY INDEX 1' THEN GOOD := FALSE;
    READINT(NLOANS,0,50,GOOD);
    IF GOOD THEN FOR I := 1 TO NLOANS DO READLINE(NAMES[I],GOOD);
    CLOSE(F); IF IORESULT <> 0 THEN GOOD := FALSE;
    IF NOT GOOD THEN BEGIN
      WRITELN('Invalid loan index. Restore your backup.'); QUIT := TRUE
    END
  END
END;

PROCEDURE SAVEIDX(T: LINE; NUMBER: INTEGER; VAR GOOD: BOOLEAN);
VAR I: INTEGER;
BEGIN
  REWRITE(F,T); GOOD := IORESULT = 0; IOERR := 0;
  IF GOOD THEN BEGIN
    SAVELN('ANNUITY INDEX 1'); SAVENUM(NUMBER);
    FOR I := 1 TO NUMBER DO SAVELN(NAMES[I]);
    IF IOERR = 0 THEN CLOSE(F,LOCK) ELSE CLOSE(F);
    IF IORESULT <> 0 THEN IOERR := 1;
    GOOD := IOERR = 0
  END
END;

PROCEDURE INDEXOUT(VAR GOOD: BOOLEAN);
BEGIN
  SAVEIDX('INDEXBAK.TEXT',NLOANS-1,GOOD);
  IF GOOD THEN SAVEIDX('LOANS.TEXT',NLOANS,GOOD)
END;

PROCEDURE NEWLOAN;
VAR GOOD: BOOLEAN; I: INTEGER;
BEGIN
  IF NLOANS = 50 THEN BEGIN WRITELN('Loan limit reached.'); EXIT(NEWLOAN) END;
  WRITE('Loan name (blank cancels): '); READLN(L.NAME);
  IF LENGTH(L.NAME) = 0 THEN EXIT(NEWLOAN);
  FOR I := 1 TO NLOANS DO IF NAMES[I] = L.NAME THEN BEGIN
    WRITELN('That name already exists.'); EXIT(NEWLOAN)
  END;
  WRITE('Description: '); READLN(L.DESCR);
  IF NOT ASKVAL('Capital: ',2,1,LIMIT,L.CAPITAL) THEN EXIT(NEWLOAN);
  IF NOT ASKVAL('Annual interest %: ',2,0,10000,L.FIXED) THEN EXIT(NEWLOAN);
  L.RATE := TRUNC(L.FIXED);
  IF NOT ASKVAL('Annual gift %: ',2,0,L.RATE,L.FIXED) THEN EXIT(NEWLOAN);
  L.GIFT := TRUNC(L.FIXED); L.FIXED := 0;
  IF NOT ASKINT('Original term in months: ',1,1200,L.TERM) THEN EXIT(NEWLOAN);
  IF NOT ASKDATE('Start DD-MM-YYYY: ',L.START) THEN EXIT(NEWLOAN);
  WRITELN('1 Constant principal  2 Interest only');
  WRITELN('3 Fixed payment including interest');
  IF NOT ASKINT('Mode: ',1,3,L.MODE) THEN EXIT(NEWLOAN);
  IF L.MODE = 3 THEN
    IF NOT ASKVAL('Monthly total: ',2,1,LIMIT,L.FIXED) THEN EXIT(NEWLOAN);
  L.COUNT := 0; SCHEDULE(FALSE,GOOD);
  IF NOT GOOD THEN EXIT(NEWLOAN);
  ACTIVE := NLOANS+1; WRITELOAN('L',GOOD);
  IF GOOD THEN BEGIN
    NLOANS := ACTIVE; NAMES[ACTIVE] := L.NAME; INDEXOUT(GOOD);
    IF GOOD THEN WRITELN('Loan created.') ELSE BEGIN
      WRITELN('Index save failed. Loan file exists.'); QUIT := TRUE
    END
  END
END;

PROCEDURE ADDEXTRA;
VAR T: PAYMENT;
BEGIN
  IF L.COUNT = 100 THEN BEGIN WRITELN('Extra payment limit reached.'); EXIT(ADDEXTRA) END;
  IF NOT ASKDATE('Payment DD-MM-YYYY: ',T.WHEN) THEN EXIT(ADDEXTRA);
  IF BEFORE(T.WHEN,L.START) THEN BEGIN
    WRITELN('Payment precedes the start.'); EXIT(ADDEXTRA)
  END;
  IF NOT ASKVAL('Extra amount: ',2,1,LIMIT,T.AMOUNT) THEN EXIT(ADDEXTRA);
  OLD := L; L.COUNT := L.COUNT+1; L.EXTRA[L.COUNT] := T; COMMIT
END;

PROCEDURE DELEXTRA;
VAR I,J: INTEGER; A,B: LINE;
BEGIN
  IF L.COUNT = 0 THEN BEGIN WRITELN('No extra payments.'); EXIT(DELEXTRA) END;
  WRITELN('Planned amounts (schedule caps them):');
  FOR I := 1 TO L.COUNT DO BEGIN
    FMTDATE(L.EXTRA[I].WHEN,A); FMT(L.EXTRA[I].AMOUNT,B);
    WRITELN(I:3,' ',A,' ',B)
  END;
  IF ASKINT('Remove number (Q cancels): ',1,L.COUNT,J) THEN BEGIN
    OLD := L;
    FOR I := J TO L.COUNT-1 DO L.EXTRA[I] := L.EXTRA[I+1];
    L.COUNT := L.COUNT-1; COMMIT
  END
END;

PROCEDURE CHGMODE;
VAR M: INTEGER; B: MONEY;
BEGIN
  WRITELN('1 Constant principal  2 Interest only');
  WRITELN('3 Fixed payment including interest');
  IF ASKINT('Mode (Q cancels): ',1,3,M) THEN BEGIN
    B := 0;
    IF M = 3 THEN
      IF NOT ASKVAL('Monthly total: ',2,1,LIMIT,B) THEN EXIT(CHGMODE);
    OLD := L; L.MODE := M; L.FIXED := B; COMMIT
  END
END;

PROCEDURE REPORT;
VAR GOOD: BOOLEAN; T: LINE;
BEGIN
  SCHEDULE(FALSE,GOOD); IF NOT GOOD THEN EXIT(REPORT);
  FNAME('R',ACTIVE,T);
  WRITE('Write/replace ',T,'? (Y/N): '); READLN(S);
  IF (S <> 'y') AND (S <> 'Y') AND (S <> 'yes') AND (S <> 'YES')
    THEN EXIT(REPORT);
  REWRITE(OUTFILE,T); GOOD := IORESULT = 0; IOERR := 0;
  IF GOOD THEN BEGIN
    TOFILE := TRUE; SCHEDULE(TRUE,GOOD); TOFILE := FALSE;
    IF GOOD AND (IOERR = 0) THEN CLOSE(OUTFILE,LOCK)
      ELSE CLOSE(OUTFILE);
    IF IORESULT <> 0 THEN IOERR := 1;
    GOOD := GOOD AND (IOERR = 0)
  END;
  IF GOOD THEN WRITELN('Report saved as ',T)
  ELSE WRITELN('Report not saved. Check disk space.')
END;

PROCEDURE LOANMENU;
VAR CHOICE: LINE;
BEGIN
  BACK := FALSE;
  REPEAT
    WRITELN; WRITELN('Loan: ',L.NAME);
    WRITELN('1 Schedule  2 Add extra  3 Remove extra');
    WRITELN('4 Text report  5 Payment mode');
    WRITELN('6 Description  7 Back  0 Quit');
    WRITE('Choice: '); READLN(CHOICE);
    IF CHOICE = '1' THEN BEGIN SCHEDULE(TRUE,OK);
      WRITE('Press Return: '); READLN(S) END;
    IF CHOICE = '2' THEN ADDEXTRA;
    IF CHOICE = '3' THEN DELEXTRA;
    IF CHOICE = '4' THEN REPORT;
    IF CHOICE = '5' THEN CHGMODE;
    IF CHOICE = '6' THEN BEGIN
      OLD := L; WRITE('Description (blank cancels, - clears): '); READLN(S);
      IF LENGTH(S) > 0 THEN BEGIN
        IF S = '-' THEN S := ''; L.DESCR := S; COMMIT
      END
    END;
    IF CHOICE = '7' THEN BACK := TRUE;
    IF CHOICE = '0' THEN QUIT := TRUE
  UNTIL BACK OR QUIT
END;

PROCEDURE MAINMENU;
VAR I: INTEGER; V: MONEY;
BEGIN
  WRITELN; WRITELN('ANNUITY - SAVED LOANS');
  FOR I := 1 TO NLOANS DO WRITELN(I:3,') ',NAMES[I]);
  WRITELN('N) New loan  0) Quit'); WRITE('Choice: '); READLN(S);
  IF (S = 'n') OR (S = 'N') THEN NEWLOAN
  ELSE IF S = '0' THEN QUIT := TRUE
  ELSE IF PARSE(S,0,V) THEN
    IF (V >= 1) AND (V <= NLOANS) THEN BEGIN
      ACTIVE := TRUNC(V); LOADLOAN(ACTIVE,OK);
      IF OK THEN LOANMENU
    END
END;

BEGIN
  LIMIT := 10000; LIMIT := LIMIT*10000*1000-1;
  QUIT := FALSE; TOFILE := FALSE; LINES := 0;
  WRITELN('ANNUITY - Apple Pascal 1.3');
  WRITELN('Use . or , for decimals; no grouping.');
  WRITELN('Q cancels numeric and date input.');
  INDEXIN;
  WHILE NOT QUIT DO MAINMENU
END.
