FUNCTION Upper;
BEGIN
  IF (c >= 'a') AND (c <= 'z') THEN
    Upper := CHR(ORD(c) - 32)
  ELSE
    Upper := c
END;

PROCEDURE ClearRow(y: INTEGER);
BEGIN
  IF (NOT rowKnown[y]) OR (LENGTH(scrRows[y]) > 0) THEN
  BEGIN
    GOTOXY(0, y);
    WRITE(CHR(29));
    scrRows[y] := '';
    rowKnown[y] := TRUE
  END
END;

PROCEDURE PutRow(y: INTEGER; s: STRING);
BEGIN
  IF (NOT rowKnown[y]) OR (scrRows[y] <> s) THEN
  BEGIN
    GOTOXY(0, y);
    WRITE(s, CHR(29));
    scrRows[y] := s;
    rowKnown[y] := TRUE
  END
END;

PROCEDURE Status;
BEGIN
  statText := s;
  PutRow(23, s)
END;

PROCEDURE ClrBelow;
VAR
  i: INTEGER;
BEGIN
  GOTOXY(0, y);
  WRITE(CHR(11));
  FOR i := y TO 23 DO
  BEGIN
    scrRows[i] := '';
    rowKnown[i] := TRUE
  END;
  IF y <= 22 THEN
    FOR i := 1 TO 19 DO lnRwOk[i] := FALSE;
  IF y <= 22 THEN
    FOR i := 6 TO 22 DO exRwOk[i] := FALSE;
  IF y <= 16 THEN sumValid := FALSE;
  Status('')
END;

PROCEDURE NewScrn;
BEGIN
  ClrBelow(1)
END;

FUNCTION MenuLine;
VAR
  c: CHAR;
  first, last, n, k, breakAt: INTEGER;
  shown: STRING[82];
  done, matched: BOOLEAN;
BEGIN
  n := LENGTH(menuText);
  first := 1;
  done := FALSE;
  REPEAT
    last := first + 76;
    IF last > n THEN
      last := n;
    IF last < n THEN
    BEGIN
      breakAt := last;
      matched := FALSE;
      WHILE (breakAt > first) AND (NOT matched) DO
      BEGIN
        IF menuText[breakAt] = ',' THEN
          matched := TRUE
        ELSE
          breakAt := breakAt - 1
      END;
      IF matched THEN last := breakAt
    END;
    shown := COPY(menuText, first, last - first + 1);
    PutRow(0, shown);
    k := LENGTH(shown) + 1;
    IF n > 77 THEN
    BEGIN
      GOTOXY(k, 0);
      WRITE(' ?');
      k := k + 2
    END;
    IF k > 79 THEN k := 79;
    GOTOXY(k, 0);
    READ(KEYBOARD, c);
    c := Upper(c);
    IF c = '?' THEN
    BEGIN
      IF last < n THEN
        first := last + 1
      ELSE
        first := 1;
      WHILE first <= n DO
        IF menuText[first] = ' ' THEN
          first := first + 1
        ELSE
          first := n + 1
    END
    ELSE
    BEGIN
      k := 1;
      matched := FALSE;
      WHILE k <= LENGTH(valid) DO
      BEGIN
        IF valid[k] = c THEN
          matched := TRUE
        ELSE
          k := k + 1;
        IF matched THEN k := LENGTH(valid) + 1
      END;
      IF matched THEN
        done := TRUE
      ELSE
        WRITE(CHR(7))
    END
  UNTIL done;
  MenuLine := c
END;

FUNCTION Leap(y: INTEGER): BOOLEAN;
BEGIN
  Leap := (y MOD 4 = 0) AND
          ((y MOD 100 <> 0) OR (y MOD 400 = 0))
END;

FUNCTION DaysMon;
BEGIN
  CASE m OF
    1,3,5,7,8,10,12: DaysMon := 31;
    4,6,9,11: DaysMon := 30;
    2:
    BEGIN
      IF Leap(y) THEN
        DaysMon := 29
      ELSE
        DaysMon := 28
    END
  END
END;

FUNCTION ValidDat(d: DateType): BOOLEAN;
VAR
  ok: BOOLEAN;
BEGIN
  ok := (d.yy >= 1) AND (d.yy <= 9999) AND
        (d.mm >= 1) AND (d.mm <= 12);
  IF ok THEN
    ok := (d.dd >= 1) AND
          (d.dd <= DaysMon(d.yy, d.mm));
  ValidDat := ok
END;

FUNCTION Serial;
VAR
  y, m: INTEGER;
  total: REAL;
BEGIN
  y := d.yy - 1;
  total := y * 365.0 + TRUNC(y / 4.0) -
           TRUNC(y / 100.0) + TRUNC(y / 400.0);
  FOR m := 1 TO d.mm - 1 DO
    total := total + DaysMon(d.yy, m);
  Serial := total + d.dd
END;

FUNCTION Later;
BEGIN
  Later := Serial(a) > Serial(b)
END;

FUNCTION SameDate;
BEGIN
  SameDate := Serial(a) = Serial(b)
END;

PROCEDURE MonthAt;
VAR
  m: REAL;
BEGIN
  a.yy := d.yy;
  a.mm := d.mm;
  m := a.yy * 12.0 + a.mm - 1 + offset;
  a.yy := TRUNC(m / 12.0);
  a.mm := TRUNC(m - TRUNC(m / 12.0) * 12.0) + 1;
  a.dd := 1;
END;

FUNCTION MaxTerm;
VAR
  yearLeft: REAL;
BEGIN
  yearLeft := 9999.0;
  yearLeft := yearLeft - d.yy;
  MaxTerm := yearLeft * 12 + 12 - d.mm
END;

PROCEDURE MonthStr;
BEGIN
  outText := '00-0000';
  outText[1] := CHR(ORD('0') + d.mm DIV 10);
  outText[2] := CHR(ORD('0') + d.mm MOD 10);
  outText[4] := CHR(ORD('0') + d.yy DIV 1000);
  outText[5] := CHR(ORD('0') + (d.yy DIV 100) MOD 10);
  outText[6] := CHR(ORD('0') + (d.yy DIV 10) MOD 10);
  outText[7] := CHR(ORD('0') + d.yy MOD 10)
END;

FUNCTION Money;
VAR
  absVal, rest, total: REAL;
  bigPart, midPart, whole, cents: INTEGER;
BEGIN
  absVal := ABS(x);
  bigPart := TRUNC(absVal / 100000000.0);
  rest := absVal - bigPart * 100000000.0;
  midPart := TRUNC(rest / 10000.0);
  rest := rest - midPart * 10000.0;
  whole := TRUNC(rest);
  cents := TRUNC((rest - whole) * 100.0 + 0.50001);
  total := bigPart * 100000000.0 +
           midPart * 10000.0 + whole + cents / 100.0;
  IF x < 0.0 THEN
    Money := -total
  ELSE
    Money := total
END;

PROCEDURE ReadLine(VAR s: STRING; maxLen: INTEGER);
VAR
  c: CHAR;
  ch: STRING[1];
  done: BOOLEAN;
BEGIN
  s := '';
  done := FALSE;
  REPEAT
    READ(c);
    IF EOLN THEN
      done := TRUE
    ELSE IF (c = CHR(8)) OR (c = CHR(127)) THEN
    BEGIN
      IF LENGTH(s) > 0 THEN
        DELETE(s, LENGTH(s), 1)
    END
    ELSE IF (c >= ' ') AND (LENGTH(s) < maxLen) THEN
    BEGIN
      ch := ' ';
      ch[1] := c;
      s := CONCAT(s, ch)
    END
  UNTIL done
END;

PROCEDURE SetField(row: INTEGER; old: STRING);
VAR
  skip: BOOLEAN;
BEGIN
  fieldRow := row;
  skip := TRUE;
  WHILE skip AND (LENGTH(old) > 0) DO
  BEGIN
    skip := old[1] = ' ';
    IF skip THEN DELETE(old, 1, 1)
  END;
  IF LENGTH(old) > 24 THEN
    DELETE(old, 25, LENGTH(old) - 24);
  fieldOld := old
END;

PROCEDURE Ask(prompt: STRING; maxLen: INTEGER; VAR outText: STRING);
VAR
  s: STRING[80];
BEGIN
  IF fieldRow > 0 THEN
  BEGIN
    GOTOXY(FieldCol, fieldRow);
    WRITE(CHR(29));
    IF (LENGTH(fieldOld) > 0) AND
       (FieldCol + LENGTH(fieldOld) + maxLen + 5 <= 79) THEN
      WRITE('(', fieldOld, ') ');
    ReadLine(s, maxLen)
  END
  ELSE
  BEGIN
    PutRow(2, prompt);
    GOTOXY(LENGTH(prompt), 2);
    ReadLine(s, maxLen);
    ClearRow(2)
  END;
  outText := s
END;

FUNCTION ParseNum(s: STRING; VAR x: REAL): BOOLEAN;
VAR
  i, digit: INTEGER;
  value, frac, place: REAL;
  dot, anyDigit, good: BOOLEAN;
BEGIN
  value := 0.0;
  frac := 0.0;
  place := 1.0;
  dot := FALSE;
  anyDigit := FALSE;
  good := TRUE;
  ParseNum := FALSE;
  FOR i := 1 TO LENGTH(s) DO
  BEGIN
    IF good THEN
    BEGIN
      IF (s[i] >= '0') AND (s[i] <= '9') THEN
      BEGIN
        digit := ORD(s[i]) - ORD('0');
        anyDigit := TRUE;
        IF dot THEN
        BEGIN
          place := place / 10.0;
          frac := frac + digit * place
        END
        ELSE IF value > 1000000000000.0 THEN
          good := FALSE
        ELSE
          value := value * 10.0 + digit
      END
      ELSE IF (s[i] = '.') OR (s[i] = ',') THEN
      BEGIN
        IF dot THEN
          good := FALSE
        ELSE
          dot := TRUE
      END
      ELSE
        good := FALSE
    END
  END;
  IF good AND anyDigit THEN
  BEGIN
    x := value + frac;
    ParseNum := TRUE
  END
END;

FUNCTION GetReal(prompt: STRING; old, minVal, maxVal: REAL;
                 cashVal: BOOLEAN): REAL;
VAR
  s: STRING[80];
  x: REAL;
  done: BOOLEAN;
BEGIN
  done := FALSE;
  x := old;
  REPEAT
    Ask(prompt, 20, s);
    IF LENGTH(s) = 0 THEN
      done := TRUE
    ELSE
    BEGIN
      IF ParseNum(s, x) AND (x >= minVal) AND (x <= maxVal) AND
         ((NOT cashVal) OR (Money(x) = x)) THEN
        done := TRUE
      ELSE
        Status('Number out of range or precision')
    END
  UNTIL done;
  GetReal := x
END;

FUNCTION ParseInt(s: STRING; VAR x: INTEGER): BOOLEAN;
VAR
  i, digit, value: INTEGER;
  good: BOOLEAN;
BEGIN
  value := 0;
  good := LENGTH(s) > 0;
  FOR i := 1 TO LENGTH(s) DO
    IF good THEN
    BEGIN
      IF (s[i] >= '0') AND (s[i] <= '9') THEN
      BEGIN
        digit := ORD(s[i]) - ORD('0');
        IF (value > 3276) OR ((value = 3276) AND (digit > 7)) THEN
          good := FALSE
        ELSE
          value := value * 10 + digit
      END
      ELSE
        good := FALSE
    END;
  IF good THEN x := value;
  ParseInt := good
END;

FUNCTION GetInt(prompt: STRING; old: INTEGER): INTEGER;
VAR
  s: STRING[80];
  x: INTEGER;
  done: BOOLEAN;
BEGIN
  done := FALSE;
  x := old;
  REPEAT
    Ask(prompt, 5, s);
    IF LENGTH(s) = 0 THEN
      done := TRUE
    ELSE
    BEGIN
      IF ParseInt(s, x) AND (x > 0) AND (x <= MaxMonth) THEN
        done := TRUE
      ELSE
        Status('Use 1 to 1200 months')
    END
  UNTIL done;
  GetInt := x
END;

PROCEDURE DateStr;
BEGIN
  outText := '0000-00-00';
  outText[1] := CHR(ORD('0') + d.yy DIV 1000);
  outText[2] := CHR(ORD('0') + (d.yy DIV 100) MOD 10);
  outText[3] := CHR(ORD('0') + (d.yy DIV 10) MOD 10);
  outText[4] := CHR(ORD('0') + d.yy MOD 10);
  outText[6] := CHR(ORD('0') + d.mm DIV 10);
  outText[7] := CHR(ORD('0') + d.mm MOD 10);
  outText[9] := CHR(ORD('0') + d.dd DIV 10);
  outText[10] := CHR(ORD('0') + d.dd MOD 10)
END;

FUNCTION ParseDat(s: STRING; VAR d: DateType): BOOLEAN;
VAR
  a, b, c: INTEGER;
BEGIN
  ParseDat := FALSE;
  IF LENGTH(s) <> 10 THEN EXIT(ParseDat);
  IF (s[5] <> '-') OR (s[8] <> '-') THEN EXIT(ParseDat);
  IF NOT ParseInt(COPY(s, 1, 4), a) THEN EXIT(ParseDat);
  IF NOT ParseInt(COPY(s, 6, 2), b) THEN EXIT(ParseDat);
  IF NOT ParseInt(COPY(s, 9, 2), c) THEN EXIT(ParseDat);
  d.yy := a;
  d.mm := b;
  d.dd := c;
  ParseDat := ValidDat(d)
END;

PROCEDURE GetDate(prompt: STRING; old: DateType; VAR outDate: DateType);
VAR
  s: STRING[80];
  done: BOOLEAN;
BEGIN
  outDate := old;
  done := FALSE;
  REPEAT
    Ask(prompt, 10, s);
    IF LENGTH(s) = 0 THEN
      done := TRUE
    ELSE IF ParseDat(s, outDate) THEN
      done := TRUE
    ELSE
      Status('Use YYYY-MM-DD')
  UNTIL done
END;

FUNCTION FindLoan(s: NameType): INTEGER;
VAR
  i: INTEGER;
BEGIN
  FindLoan := 0;
  FOR i := 1 TO loanCnt DO
    IF loans[i].name = s THEN
      FindLoan := i
END;

PROCEDURE LoadData;
BEGIN
  loanCnt := 0;
  extCount := 0;
  (*$I-*)
  RESET(lnFile, DataFNam);
  (*$I+*)
  IF IORESULT <> 0 THEN
    EXIT(LoadData);
  WHILE NOT EOF(lnFile) AND (loanCnt < MaxLoans) DO
  BEGIN
    loanCnt := loanCnt + 1;
    loans[loanCnt] := lnFile^;
    GET(lnFile)
  END;
  CLOSE(lnFile);
  (*$I-*)
  RESET(exFile, ExtraFNm);
  (*$I+*)
  IF IORESULT <> 0 THEN
    EXIT(LoadData);
  WHILE NOT EOF(exFile) AND (extCount < MaxExtra) DO
  BEGIN
    extCount := extCount + 1;
    extras[extCount] := exFile^;
    GET(exFile)
  END;
  CLOSE(exFile)
END;

PROCEDURE SaveLoan;
VAR
  i: INTEGER;
BEGIN
  REWRITE(lnFile, DataFNam);
  FOR i := 1 TO loanCnt DO
  BEGIN
    lnFile^ := loans[i];
    PUT(lnFile)
  END;
  CLOSE(lnFile, LOCK)
END;

PROCEDURE SaveExtr;
VAR
  i: INTEGER;
BEGIN
  REWRITE(exFile, ExtraFNm);
  FOR i := 1 TO extCount DO
  BEGIN
    exFile^ := extras[i];
    PUT(exFile)
  END;
  CLOSE(exFile, LOCK)
END;

PROCEDURE LoanSum(l: LoanType);
VAR
  s: STRING[80];
  payment: REAL;
BEGIN
  PutRow(7, CONCAT(' Name...................: ', l.name));
  PutRow(8, CONCAT(' Title..................: ', l.title));
  IF (NOT sumValid) OR (sumLoan.loanAmt <> l.loanAmt) THEN
  BEGIN
    PutRow(9, ' Loan Amount............:');
    GOTOXY(FieldCol, 9);
    WRITE(l.loanAmt:12:2)
  END;
  IF (NOT sumValid) OR (sumLoan.months <> l.months) THEN
  BEGIN
    PutRow(10, ' Duration...............:');
    GOTOXY(FieldCol, 10);
    WRITE(l.months:6)
  END;
  IF (NOT sumValid) OR (sumLoan.rate <> l.rate) THEN
  BEGIN
    PutRow(11, ' Interest Rate %........:');
    GOTOXY(FieldCol, 11);
    WRITE(l.rate:8:3)
  END;
  IF (NOT sumValid) OR (sumLoan.rebate <> l.rebate) THEN
  BEGIN
    PutRow(12, ' Rebate %...............:');
    GOTOXY(FieldCol, 12);
    WRITE(l.rebate:8:3)
  END;
  IF l.fixPay > 0.0 THEN
    payment := l.fixPay
  ELSE IF l.freePay THEN
    payment := 0.0
  ELSE IF l.months > 0 THEN
    payment := Money(l.loanAmt / l.months)
  ELSE
    payment := 0.0;
  IF (NOT sumValid) OR (sumLoan.loanAmt <> l.loanAmt) OR
     (sumLoan.months <> l.months) OR
     (sumLoan.freePay <> l.freePay) OR
     (sumLoan.fixPay <> l.fixPay) THEN
  BEGIN
    PutRow(13, ' Monthly Payment........:');
    GOTOXY(FieldCol, 13);
    WRITE(payment:12:2)
  END;
  IF (NOT sumValid) OR (sumLoan.freePay <> l.freePay) THEN
    IF l.freePay THEN
      PutRow(14, ' No repayment...........: On')
    ELSE
      PutRow(14, ' No repayment...........: Off');
  IF l.fixPay > 0.0 THEN
  BEGIN
    IF (NOT sumValid) OR
       (sumLoan.fixPay <> l.fixPay) THEN
      PutRow(15, ' Fixed monthly amount...: On')
  END
  ELSE IF (NOT sumValid) OR
          (sumLoan.fixPay <> l.fixPay) THEN
    PutRow(15, ' Fixed monthly amount...: Off');
  IF (NOT sumValid) OR (sumLoan.starts.yy <> l.starts.yy) OR
     (sumLoan.starts.mm <> l.starts.mm) OR
     (sumLoan.starts.dd <> l.starts.dd) THEN
  BEGIN
    DateStr(l.starts, s);
    PutRow(16, CONCAT(' Start Date.............: ', s))
  END;
  sumLoan := l;
  sumValid := TRUE
END;

PROCEDURE MainScrn;
VAR
  i: INTEGER;
BEGIN
  FOR i := 1 TO 6 DO ClearRow(i);
  FOR i := 17 TO 22 DO ClearRow(i);
  IF current > 0 THEN
    LoanSum(loans[current])
  ELSE
  BEGIN
    PutRow(7, ' No loan selected');
    FOR i := 8 TO 16 DO ClearRow(i);
    sumValid := FALSE
  END
END;

PROCEDURE ListLoan;
VAR
  i, firstLn, lastLoan, row: INTEGER;
  pageText: STRING[40];
  index: INTEGER;
  changed: BOOLEAN;
BEGIN
  firstLn := loanPage * 19 + 1;
  lastLoan := firstLn + 18;
  IF lastLoan > loanCnt THEN lastLoan := loanCnt;
  FOR row := 4 TO 22 DO
  BEGIN
    index := loanPage * 19 + row - 3;
    IF index > lastLoan THEN index := 0;
    changed := NOT lnRwOk[row - 3];
    IF NOT changed THEN
      changed := lnRwIdx[row - 3] <> index;
    IF (NOT changed) AND (index > 0) THEN
      changed := loanRows[row - 3] <> loans[index].name;
    IF changed THEN
    BEGIN
      IF index > 0 THEN
      BEGIN
        GOTOXY(0, row);
        WRITE(' ', index:2, '. ', loans[index].name, CHR(29));
        loanRows[row - 3] := loans[index].name;
        scrRows[row] := 'X';
        rowKnown[row] := TRUE
      END
      ELSE
      BEGIN
        IF lnRwIdx[row - 3] > 0 THEN
        BEGIN
          GOTOXY(0, row);
          WRITE(CHR(29));
          scrRows[row] := '';
          rowKnown[row] := TRUE
        END
      END;
      lnRwIdx[row - 3] := index;
      lnRwOk[row - 3] := TRUE
    END
  END;
  IF loanCnt > 19 THEN
  BEGIN
    pageText := 'Page ';
    IF loanPage = 0 THEN
      pageText := CONCAT(pageText, '1')
    ELSE
      pageText := CONCAT(pageText, '2');
    PutRow(2, CONCAT(pageText, ' - [ previous, ] next'))
  END
  ELSE
    ClearRow(2)
END;

FUNCTION ChooseLn(prompt: STRING): INTEGER;
VAR
  s: STRING[80];
  n: INTEGER;
BEGIN
  n := 0;
  Ask(prompt, 5, s);
  IF LENGTH(s) > 0 THEN
  BEGIN
    IF (NOT ParseInt(s, n)) OR (n < 1) OR (n > loanCnt) THEN
    BEGIN
      Status('Invalid loan number');
      n := 0
    END
  END;
  ChooseLn := n
END;

PROCEDURE NewLoan;
VAR
  l: LoanType;
  c: CHAR;
  s: STRING[80];
  saved: BOOLEAN;
BEGIN
  l.name := '';
  l.title := '';
  l.loanAmt := 0.0;
  l.rate := 0.0;
  l.rebate := 0.0;
  l.months := 12;
  l.freePay := FALSE;
  l.fixPay := 0.0;
  l.starts.yy := 2026;
  l.starts.mm := 1;
  l.starts.dd := 1;
  saved := FALSE;
  ClrBelow(2);
  REPEAT
    PutRow(2, CONCAT(' Name...................: ', l.name));
    PutRow(3, CONCAT(' Title..................: ', l.title));
    c := MenuLine('New Loan: N(ame, T(itle, S(ave, Q(uit', 'NTSQ');
    CASE c OF
      'N':
      BEGIN
        SetField(2, l.name);
        Ask('Name: ', NameSize, s);
        fieldRow := 0;
        IF LENGTH(s) > NameSize THEN
          DELETE(s, NameSize + 1, LENGTH(s) - NameSize);
        IF LENGTH(s) > 0 THEN
          l.name := s
      END;
      'T':
      BEGIN
        SetField(3, l.title);
        Ask('Title: ', TitlSiz, s);
        fieldRow := 0;
        IF LENGTH(s) > TitlSiz THEN
          DELETE(s, TitlSiz + 1, LENGTH(s) - TitlSiz);
        IF LENGTH(s) > 0 THEN
          l.title := s
      END;
      'S':
      BEGIN
        IF (l.name = '') OR (FindLoan(l.name) <> 0) THEN
          Status('Name missing or already exists')
        ELSE IF loanCnt >= MaxLoans THEN
          Status('Loan storage full')
        ELSE
        BEGIN
          loanCnt := loanCnt + 1;
          loans[loanCnt] := l;
          SaveLoan;
          current := loanCnt;
          Status('New loan saved');
          saved := TRUE
        END
      END
    END
{ UNTIL (c = 'Q') OR saved --> only "Q" exits menu }
  UNTIL (c = 'Q')
END;

PROCEDURE RemLoan;
VAR
  n, i, j: INTEGER;
  name: NameType;
  c: CHAR;
BEGIN
  n := ChooseLn('Remove number (Enter cancels): ');
  IF n = 0 THEN EXIT(RemLoan);
  name := loans[n].name;
  PutRow(2, CONCAT('Remove ', name, '? Y/N'));
  READ(KEYBOARD, c);
  ClearRow(2);
  IF Upper(c) <> 'Y' THEN EXIT(RemLoan);
  FOR i := n TO loanCnt - 1 DO
    loans[i] := loans[i + 1];
  loanCnt := loanCnt - 1;
  j := 1;
  FOR i := 1 TO extCount DO
  BEGIN
    IF extras[i].loanName <> name THEN
    BEGIN
      extras[j] := extras[i];
      j := j + 1
    END
  END;
  extCount := j - 1;
  IF current = n THEN current := 0;
  IF current > n THEN current := current - 1;
  SaveLoan;
  SaveExtr;
  Status('Loan removed')
END;

PROCEDURE LoanMenu;
VAR
  c: CHAR;
  n: INTEGER;
BEGIN
  loanPage := 0;
  NewScrn;
  ListLoan;
  REPEAT
    c := MenuLine(
      'Loans: S(elect, N(ew), R(emove), [previous, ]next, Q(uit',
      'SNRQ[]');
    CASE c OF
      'S':
      BEGIN
        n := ChooseLn('Select number (Enter cancels): ');
        IF n > 0 THEN
        BEGIN
          current := n;
          Status('Loan selected')
        END
      END;
      'N':
      BEGIN
        NewLoan;
      END;
      'R': RemLoan;
      '[':
        IF loanPage > 0 THEN loanPage := loanPage - 1;
      ']':
        IF (loanPage + 1) * 19 < loanCnt THEN
          loanPage := loanPage + 1
    END;
    IF loanCnt = 0 THEN
      loanPage := 0
    ELSE IF loanPage * 19 >= loanCnt THEN
      loanPage := (loanCnt - 1) DIV 19;
    IF c <> 'Q' THEN
    BEGIN
      ListLoan
    END
  UNTIL c = 'Q'
END;

PROCEDURE DataMenu;
VAR
  c: CHAR;
  s: STRING[80];
  menuTxt: Str255;
  done: BOOLEAN;
BEGIN
  IF current = 0 THEN
  BEGIN
    Status('Select a loan first');
    EXIT(DataMenu);
  END;
  workLoan := loans[current];
  dirty := FALSE;
  done := FALSE;
  REPEAT
    LoanSum(workLoan);
    menuTxt := 'Data: L(oan, D(uration, I(nterest, R(ebate, ';
    INSERT('T(itle, B(egin date, F(ix payment, N(o repayment, ',
           menuTxt, LENGTH(menuTxt) + 1);
    INSERT('S(ave, Q(uit', menuTxt, LENGTH(menuTxt) + 1);
    c := MenuLine(menuTxt, 'LDIRTB FNSQ');
    CASE c OF
      'L':
      BEGIN
        SetField(9, '');
        workLoan.loanAmt :=
          GetReal('Loan amount: ', workLoan.loanAmt, 0.01,
                  MaxAmt, TRUE)
      END;
      'D':
      BEGIN
        SetField(10, '');
        workLoan.months :=
          GetInt('Duration months: ', workLoan.months)
      END;
      'I':
      BEGIN
        SetField(11, '');
        workLoan.rate :=
          GetReal('Interest percent: ', workLoan.rate, 0.0,
                  100.0, FALSE)
      END;
      'R':
      BEGIN
        SetField(12, '');
        workLoan.rebate :=
          GetReal('Rebate percent: ', workLoan.rebate, 0.0,
                  workLoan.rate, FALSE)
      END;
      'T':
      BEGIN
        SetField(8, workLoan.title);
        Ask('Title: ', TitlSiz, s);
        IF LENGTH(s) > TitlSiz THEN
          DELETE(s, TitlSiz + 1, LENGTH(s) - TitlSiz);
        IF LENGTH(s) > 0 THEN
          workLoan.title := s
      END;
      'B':
      BEGIN
        DateStr(workLoan.starts, s);
        SetField(16, s);
        GetDate('Start YYYY-MM-DD: ', workLoan.starts,
                workLoan.starts)
      END;
      'F':
      BEGIN
        SetField(15, '');
        workLoan.freePay := FALSE;
        workLoan.fixPay :=
          GetReal('Monthly payment incl. interest: ',
                  workLoan.fixPay, 0.01, MaxAmt, TRUE)
      END;
      'N':
      BEGIN
        IF workLoan.freePay THEN
          workLoan.freePay := FALSE
        ELSE
        BEGIN
          workLoan.freePay := TRUE;
          workLoan.fixPay := 0.0
        END
      END;
      'S':
      BEGIN
        IF (workLoan.loanAmt <= 0.0) OR
           (workLoan.loanAmt > MaxAmt) OR
           (workLoan.months <= 0) OR
           (workLoan.months > MaxMonth) OR
           (NOT ValidDat(workLoan.starts)) OR
           (workLoan.starts.yy +
            (workLoan.starts.mm - 1 + workLoan.months) DIV 12 > 9999) OR
           (workLoan.rate > 100.0) OR
           (workLoan.rebate < 0.0) OR
           (workLoan.rebate > workLoan.rate) OR
           (workLoan.fixPay < 0.0) OR
           (workLoan.fixPay > MaxAmt) OR
           ((workLoan.fixPay > 0.0) AND workLoan.freePay) THEN
          Status('Check loan values and payment options')
        ELSE
        BEGIN
          Status('Saving data');
          loans[current] := workLoan;
          SaveLoan;
          dirty := FALSE;
          Status('Data saved')
        END
      END;
      'Q': done := TRUE
    END;
    fieldRow := 0;
    IF (c <> 'S') AND (c <> 'Q') THEN dirty := TRUE
  UNTIL done
END;

PROCEDURE ListExtr;
VAR
  i, row: INTEGER;
  dateText: STRING;
  changed: BOOLEAN;
BEGIN
  row := 6;
  FOR i := 1 TO extCount DO
    IF (current > 0) AND
       (extras[i].loanName = loans[current].name) AND
       (row <= 22) THEN
    BEGIN
      changed := NOT exRwOk[row];
      IF NOT changed THEN
        changed := exRwIdx[row] <> i;
      IF NOT changed THEN
        changed := exRwDat[row].yy <> extras[i].paid.yy;
      IF NOT changed THEN
        changed := exRwDat[row].mm <> extras[i].paid.mm;
      IF NOT changed THEN
        changed := exRwDat[row].dd <> extras[i].paid.dd;
      IF NOT changed THEN
        changed := exRwAmt[row] <> extras[i].amount;
      IF changed THEN
      BEGIN
        DateStr(extras[i].paid, dateText);
        GOTOXY(0, row);
        WRITE(i:3, ' ', dateText);
        GOTOXY(20, row);
        WRITE(extras[i].amount:12:2, CHR(29));
        scrRows[row] := 'X';
        rowKnown[row] := TRUE;
        exRwIdx[row] := i;
        exRwDat[row] := extras[i].paid;
        exRwAmt[row] := extras[i].amount;
        exRwOk[row] := TRUE
      END;
      row := row + 1
    END;
  WHILE row <= 22 DO
  BEGIN
    IF exRwOk[row] AND (exRwIdx[row] > 0) THEN
    BEGIN
      ClearRow(row);
      exRwIdx[row] := 0
    END;
    exRwOk[row] := TRUE;
    row := row + 1
  END
END;

PROCEDURE AddExtra;
VAR
  e: ExtRec;
  endDate: DateType;
BEGIN
  IF (loans[current].loanAmt <= 0.0) OR
     (loans[current].months <= 0) THEN
  BEGIN
    Status('Complete loan data first');
    EXIT(AddExtra)
  END;
  IF extCount >= MaxExtra THEN
  BEGIN
    Status('Extra payment storage full');
    EXIT(AddExtra)
  END;
  e.loanName := loans[current].name;
  GetDate('Payment YYYY-MM-DD: ', loans[current].starts, e.paid);
  IF loans[current].fixPay > 0.0 THEN
    MonthAt(loans[current].starts, MaxTerm(loans[current].starts),
            endDate)
  ELSE
    MonthAt(loans[current].starts, loans[current].months, endDate);
  IF Later(loans[current].starts, e.paid) OR
     (NOT Later(endDate, e.paid)) THEN
  BEGIN
    Status('Payment date is outside loan term');
    EXIT(AddExtra)
  END;
  e.amount := GetReal('Extra amount: ', 0.0, 0.01, MaxAmt, TRUE);
  IF e.amount <= 0.0 THEN EXIT(AddExtra);
  extCount := extCount + 1;
  extras[extCount] := e;
  SaveExtr;
  Status('Extra payment saved')
END;

PROCEDURE RemExtr;
VAR
  s: STRING[80];
  n, i: INTEGER;
BEGIN
  Ask('Extra number (Enter cancels): ', 5, s);
  IF LENGTH(s) = 0 THEN EXIT(RemExtr);
  IF (NOT ParseInt(s, n)) OR (n < 1) OR (n > extCount) THEN
  BEGIN
    Status('Invalid extra number');
    EXIT(RemExtr)
  END;
  IF extras[n].loanName <> loans[current].name THEN
  BEGIN
    Status('Extra belongs to another loan');
    EXIT(RemExtr)
  END;
  FOR i := n TO extCount - 1 DO
    extras[i] := extras[i + 1];
  extCount := extCount - 1;
  SaveExtr;
  Status('Extra payment removed')
END;

END;
