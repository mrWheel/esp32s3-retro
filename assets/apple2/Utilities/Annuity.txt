10 REM ANNUITY CALCULATOR
20 REM INPUT: PRINCIPAL, TERM, RATE
30 REM OUTPUT: MONTHLY PAYMENT, TOTALS
100 HOME
110 PRINT "ANNUITY CALCULATOR"
120 PRINT "------------------"
130 PRINT
140 REM --- INPUT ---
150 INPUT "PRINCIPAL: ";P
160 IF P <= 0 THEN PRINT "MUST BE GREATER THAN 0": GOTO 150
170 INPUT "TERM IN YEARS: ";Y
180 IF Y <= 0 THEN PRINT "MUST BE GREATER THAN 0": GOTO 170
190 INPUT "ANNUAL INTEREST RATE (%): ";R
200 IF R < 0 THEN PRINT "MUST NOT BE NEGATIVE": GOTO 190
300 REM --- CALCULATION ---
310 N = INT(Y * 12 + 0.5): REM NUMBER OF MONTHLY PAYMENTS
320 I = R / 100 / 12: REM MONTHLY INTEREST RATE
330 IF I = 0 THEN A = P / N: GOTO 360: REM NO INTEREST
340 F = (1 + I) ^ N
350 A = P * I * F / (F - 1): REM PAYMENT = P*I / (1-(1+I)^-N)
360 A = INT(A * 100 + 0.5) / 100: REM ROUND TO CENTS
370 T = A * N: REM TOTAL PAID
380 C = T - P: REM TOTAL INTEREST
400 REM --- OUTPUT ---
410 PRINT
420 PRINT "PRINCIPAL:       ";: GOSUB 600: PRINT P
430 PRINT "TERM:            ";Y;" YEARS (";N;" MONTHS)"
440 PRINT "ANNUAL RATE:     ";R;"%"
450 PRINT
460 PRINT "MONTHLY PAYMENT: ";: GOSUB 600: PRINT A
470 PRINT "TOTAL PAID:      ";: GOSUB 600: PRINT INT(T * 100 + 0.5) / 100
480 PRINT "TOTAL INTEREST:  ";: GOSUB 600: PRINT INT(C * 100 + 0.5) / 100
490 PRINT
500 INPUT "ANOTHER CALCULATION (Y/N)? ";A$
510 IF A$ = "Y" THEN 100
520 END
600 REM --- PRINT CURRENCY MARKER ---
610 PRINT "$";
620 RETURN
