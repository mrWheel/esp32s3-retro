PROGRAM BENCHUC;
CONST
  INTRNDS = 35;
  INTMOD = 100;
  INTSUM = 5050;
  SIEVEN = 500;
  SIEVERNDS = 1;
  SIEVEPRIMES = 95;
  INTMODEL = 5128;
  SIEVEMODEL = 5020;
  CAPADDR = -16224;
TYPE
  BYTEARR = PACKED ARRAY[0..255] OF CHAR;
  PTRREC = RECORD
    CASE BOOLEAN OF
      TRUE: (A: INTEGER);
      FALSE: (P: ^BYTEARR)
  END;
VAR
  PT: PTRREC;
  STARTT, DOTLAST, ELAPSED, MODEL, INDEX: REAL;
  SCALE, ROUNDNO, ITEMNO, EXPECT, X, COUNT: INTEGER;
  DOTS: BOOLEAN;
  FLAGS: PACKED ARRAY[0..SIEVEN] OF BOOLEAN;

PROCEDURE PRINTNUM(R: REAL);
VAR HI, LO: INTEGER;
BEGIN
  HI := TRUNC(R / 10000.0);
  LO := TRUNC(R - HI * 10000.0);
  IF HI > 0 THEN
  BEGIN
    WRITE(HI);
    IF LO < 1000 THEN WRITE('0');
    IF LO < 100 THEN WRITE('0');
    IF LO < 10 THEN WRITE('0')
  END;
  WRITE(LO)
END;

FUNCTION NOW: REAL;
VAR T: REAL;
BEGIN
  T := ORD(PT.P^[1]);
  T := T + ORD(PT.P^[2]) * 256.0;
  T := T + ORD(PT.P^[3]) * 65536.0;
  NOW := T + ORD(PT.P^[4]) * 16777216.0
END;

PROCEDURE STARTTIMER;
BEGIN
  STARTT := NOW;
  DOTLAST := STARTT;
  DOTS := FALSE
END;

PROCEDURE DOTCHECK;
VAR T: REAL;
BEGIN
  T := NOW;
  IF T - DOTLAST >= 5000.0 THEN
  BEGIN
    DOTLAST := T;
    DOTS := TRUE;
    WRITE('.')
  END
END;

PROCEDURE STOPTIMER;
BEGIN
  ELAPSED := NOW - STARTT;
  IF DOTS THEN
    WRITELN;
  DOTS := FALSE
END;

PROCEDURE REPORT(MODELMS: INTEGER);
BEGIN
  MODEL := MODELMS;
  MODEL := MODEL * SCALE;
  WRITE('  Elapsed: ');
  PRINTNUM(ELAPSED);
  WRITELN(' ms');
  WRITE('  P-CODE MODEL ONLY (not measured): 1.023MHz=');
  PRINTNUM(MODEL);
  WRITELN(' ms');
  IF ELAPSED < 1.0 THEN
    WRITELN('  Index unavailable: elapsed <1 ms. Increase scale.')
  ELSE
  BEGIN
    INDEX := MODEL * 100.0 / ELAPSED;
    IF INDEX > 30000.0 THEN
      INDEX := 30000.0;
    WRITELN('  Index vs 1.023MHz MODEL: ', TRUNC(INDEX + 0.5),
            ' (100=model, higher=faster; theoretical)')
  END
END;

BEGIN
  PT.A := CAPADDR;
  IF ORD(PT.P^[0]) <> 193 THEN
    WRITELN('Timer unavailable. Update firmware.')
  ELSE
  BEGIN
    REPEAT
      WRITE('SCALE (1-9)? ');
      READLN(SCALE)
    UNTIL (SCALE >= 1) AND (SCALE <= 9);
    WRITELN;
    WRITELN('BENCHUC v2 SCALE=', SCALE);

    WRITELN('@BEGIN INTEGER');
    STARTTIMER;
    X := 1;
    FOR ROUNDNO := 1 TO INTRNDS * SCALE DO
    BEGIN
      FOR ITEMNO := 1 TO INTMOD DO
      BEGIN
        X := X + ITEMNO;
        IF X > 30000 THEN
          X := X - 30000
      END;
      DOTCHECK
    END;
    STOPTIMER;
    EXPECT := 1;
    FOR ROUNDNO := 1 TO INTRNDS * SCALE DO
      IF EXPECT > 30000 - INTSUM THEN
        EXPECT := EXPECT - (30000 - INTSUM)
      ELSE
        EXPECT := EXPECT + INTSUM;
    IF X = EXPECT THEN
      WRITELN('@END INTEGER PASS X=', X)
    ELSE
      WRITELN('@END INTEGER FAIL X=', X, ' EXPECTED=', EXPECT);
    REPORT(INTMODEL);

    WRITELN('@BEGIN SIEVE');
    STARTTIMER;
    FOR ROUNDNO := 1 TO SIEVERNDS * SCALE DO
    BEGIN
      FOR ITEMNO := 0 TO SIEVEN DO
        FLAGS[ITEMNO] := TRUE;
      DOTCHECK;
      FOR ITEMNO := 2 TO SIEVEN DIV 2 DO
        IF FLAGS[ITEMNO] THEN
        BEGIN
          X := ITEMNO + ITEMNO;
          WHILE X <= SIEVEN DO
          BEGIN
            FLAGS[X] := FALSE;
            X := X + ITEMNO
          END
        END;
      DOTCHECK;
      COUNT := 0;
      FOR ITEMNO := 2 TO SIEVEN DO
        IF FLAGS[ITEMNO] THEN
          COUNT := COUNT + 1
    END;
    STOPTIMER;
    IF COUNT = SIEVEPRIMES THEN
      WRITELN('@END SIEVE PASS PRIMES=', COUNT)
    ELSE
      WRITELN('@END SIEVE FAIL PRIMES=', COUNT);
    REPORT(SIEVEMODEL);
    IF COUNT = SIEVEPRIMES THEN
      WRITELN('@DONE PASS')
    ELSE
      WRITELN('@DONE FAIL')
  END
END.
